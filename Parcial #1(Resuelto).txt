package pa.ac.unachi.rrhh.datos;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;
import java.util.ArrayList;
import java.util.List;
import pa.ac.unachi.rrhh.modelo.Departamento;

/**
 * Acceso a datos de la tabla departamento. No valida reglas de negocio.
 */
public class DepartamentoDAO {

    /** Inserta el departamento y devuelve el id generado. */
    public int insertar(Departamento d) throws SQLException {
        String sql = "INSERT INTO departamento (codigo, nombre, salario_base, dotacion_maxima) VALUES (?, ?, ?, ?)";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
            ps.setString(1, d.getCodigo());
            ps.setString(2, d.getNombre());
            ps.setBigDecimal(3, d.getSalarioBase());
            ps.setInt(4, d.getDotacionMaxima());
            ps.executeUpdate();
            try (ResultSet rs = ps.getGeneratedKeys()) {
                rs.next();
                return rs.getInt(1);
            }
        }
    }

    public List<Departamento> listar() throws SQLException {
        String sql = "SELECT id, codigo, nombre, salario_base, dotacion_maxima FROM departamento ORDER BY codigo";
        List<Departamento> lista = new ArrayList<>();
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                lista.add(mapearDepartamento(rs));
            }
        }
        return lista;
    }

    public Departamento buscarPorId(int id) throws SQLException {
        String sql = "SELECT id, codigo, nombre, salario_base, dotacion_maxima FROM departamento WHERE id = ?";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setInt(1, id);
            try (ResultSet rs = ps.executeQuery()) {
                return rs.next() ? mapearDepartamento(rs) : null;
            }
        }
    }

    public void actualizar(Departamento d) throws SQLException {
        String sql = "UPDATE departamento SET codigo = ?, nombre = ?, salario_base = ?, dotacion_maxima = ? WHERE id = ?";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setString(1, d.getCodigo());
            ps.setString(2, d.getNombre());
            ps.setBigDecimal(3, d.getSalarioBase());
            ps.setInt(4, d.getDotacionMaxima());
            ps.setInt(5, d.getId());
            ps.executeUpdate();
        }
    }

    public void eliminar(int id) throws SQLException {
        String sql = "DELETE FROM departamento WHERE id = ?";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setInt(1, id);
            ps.executeUpdate();
        }
    }

    private Departamento mapearDepartamento(ResultSet rs) throws SQLException {
        Departamento d = new Departamento();
        d.setId(rs.getInt("id"));
        d.setCodigo(rs.getString("codigo"));
        d.setNombre(rs.getString("nombre"));
        d.setSalarioBase(rs.getBigDecimal("salario_base"));
        d.setDotacionMaxima(rs.getInt("dotacion_maxima"));
        return d;
    }
}

package pa.ac.unachi.rrhh.datos;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;
import pa.ac.unachi.rrhh.modelo.Dotacion;

/**
 * Acceso a datos de la tabla dotacion. No valida reglas de negocio.
 */
public class DotacionDAO {

    /** Crea el registro de dotación (cantidad 0) de un departamento nuevo. */
    public void crearRegistro(int departamentoId) throws SQLException {
        String sql = "INSERT INTO dotacion (departamento_id, cantidad) VALUES (?, 0)";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setInt(1, departamentoId);
            ps.executeUpdate();
        }
    }

    public List<Dotacion> listar() throws SQLException {
        String sql = "SELECT t.id, t.departamento_id, t.cantidad, d.codigo, d.nombre, d.dotacion_maxima "
                + "FROM dotacion t JOIN departamento d ON d.id = t.departamento_id "
                + "ORDER BY d.codigo";
        List<Dotacion> lista = new ArrayList<>();
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                lista.add(mapearDotacion(rs));
            }
        }
        return lista;
    }

    public Dotacion buscarPorDepartamento(int departamentoId) throws SQLException {
        String sql = "SELECT t.id, t.departamento_id, t.cantidad, d.codigo, d.nombre, d.dotacion_maxima "
                + "FROM dotacion t JOIN departamento d ON d.id = t.departamento_id "
                + "WHERE t.departamento_id = ?";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setInt(1, departamentoId);
            try (ResultSet rs = ps.executeQuery()) {
                return rs.next() ? mapearDotacion(rs) : null;
            }
        }
    }

    public void actualizarCantidad(int departamentoId, int cantidad) throws SQLException {
        String sql = "UPDATE dotacion SET cantidad = ? WHERE departamento_id = ?";
        try (Connection con = Conexion.obtener();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setInt(1, cantidad);
            ps.setInt(2, departamentoId);
            ps.executeUpdate();
        }
    }

    private Dotacion mapearDotacion(ResultSet rs) throws SQLException {
        Dotacion t = new Dotacion();
        t.setId(rs.getInt("id"));
        t.setDepartamentoId(rs.getInt("departamento_id"));
        t.setCantidad(rs.getInt("cantidad"));
        t.setCodigoDepartamento(rs.getString("codigo"));
        t.setNombreDepartamento(rs.getString("nombre"));
        t.setDotacionMaximaDepartamento(rs.getInt("dotacion_maxima"));
        return t;
    }
}

package pa.ac.unachi.rrhh.negocio;

import java.math.BigDecimal;
import java.sql.SQLException;
import java.sql.SQLIntegrityConstraintViolationException;
import java.util.List;
import pa.ac.unachi.rrhh.datos.DepartamentoDAO;
import pa.ac.unachi.rrhh.datos.DotacionDAO;
import pa.ac.unachi.rrhh.modelo.Departamento;

/**
 * Reglas de negocio de los departamentos.
 */
public class DepartamentoService {

    private final DepartamentoDAO departamentoDAO = new DepartamentoDAO();
    private final DotacionDAO dotacionDAO = new DotacionDAO();

    public List<Departamento> listar() throws SQLException {
        return departamentoDAO.listar();
    }

    public Departamento buscar(int id) throws NegocioException, SQLException {
        Departamento d = departamentoDAO.buscarPorId(id);
        if (d == null) {
            throw new NegocioException("El departamento solicitado no existe.");
        }
        return d;
    }

    public void crear(Departamento d) throws NegocioException, SQLException {
        validar(d);
        try {
            int id = departamentoDAO.insertar(d);
            dotacionDAO.crearRegistro(id);
        } catch (SQLIntegrityConstraintViolationException e) {
            throw new NegocioException("Ya existe un departamento con el código " + d.getCodigo() + ".");
        }
    }

    public void actualizar(Departamento d) throws NegocioException, SQLException {
        validar(d);
        try {
            departamentoDAO.actualizar(d);
        } catch (SQLIntegrityConstraintViolationException e) {
            throw new NegocioException("Ya existe un departamento con el código " + d.getCodigo() + ".");
        }
    }

    public void eliminar(int id) throws SQLException {
        departamentoDAO.eliminar(id);
    }

    private void validar(Departamento d) throws NegocioException {
        if (d.getCodigo() == null || d.getCodigo().isBlank()) {
            throw new NegocioException("El código es obligatorio.");
        }
        if (!d.getCodigo().matches("DEP-\\d{3}")) {
            throw new NegocioException("El código debe tener el formato DEP-000.");
        }
        if (d.getNombre() == null || d.getNombre().isBlank()) {
            throw new NegocioException("El nombre es obligatorio.");
        }
        if (d.getNombre().length() > 100) {
            throw new NegocioException("El nombre no puede superar los 100 caracteres.");
        }

        // Mensajes exactos exigidos por la pauta del examen (Casos de prueba B2 y B3)
        if (d.getSalarioBase() == null || d.getSalarioBase().compareTo(BigDecimal.ZERO) <= 0) {
            throw new NegocioException("El salario base debe ser mayor a cero.");
        }
        if (d.getDotacionMaxima() < 1) {
            throw new NegocioException("La dotación máxima debe ser de al menos 1 colaborador.");
        }
    }
}

Java
package pa.ac.unachi.rrhh.negocio;

import java.sql.SQLException;
import java.util.List;
import pa.ac.unachi.rrhh.datos.DotacionDAO;
import pa.ac.unachi.rrhh.modelo.Dotacion;

/**
 * Reglas de negocio de la dotación de personal: altas y bajas.
 */
public class DotacionService {

    private final DotacionDAO dotacionDAO = new DotacionDAO();

    public List<Dotacion> listar() throws SQLException {
        return dotacionDAO.listar();
    }

    /** Alta: ingresan colaboradores al departamento (contratación o traslado). */
    public void registrarAlta(int departamentoId, int cantidad) throws NegocioException, SQLException {
        if (cantidad <= 0) {
            throw new NegocioException("La cantidad de colaboradores debe ser mayor a cero.");
        }
        Dotacion actual = buscar(departamentoId);
        int nueva = actual.getCantidad() + cantidad;

        // Regla Tema B (Casos de prueba B5 y B6): Validar dotación máxima
        if (nueva > actual.getDotacionMaximaDepartamento()) {
            int vacantesDisponibles = actual.getDotacionMaximaDepartamento() - actual.getCantidad();
            throw new NegocioException("No se puede contratar a " + cantidad + " colaboradores en " 
                    + actual.getNombreDepartamento() + ". Solo puede contratar " 
                    + Math.max(0, vacantesDisponibles) + " colaboradores más.");
        }

        dotacionDAO.actualizarCantidad(departamentoId, nueva);
    }

    /** Baja: salen colaboradores del departamento (renuncia, despido o traslado). */
    public void registrarBaja(int departamentoId, int cantidad) throws NegocioException, SQLException {
        if (cantidad <= 0) {
            throw new NegocioException("La cantidad de colaboradores debe ser mayor a cero.");
        }
        Dotacion actual = buscar(departamentoId);
        if (cantidad > actual.getCantidad()) {
            throw new NegocioException("No se puede dar de baja a " + cantidad + " colaboradores: "
                    + actual.getNombreDepartamento() + " solo tiene " + actual.getCantidad() + ".");
        }
        dotacionDAO.actualizarCantidad(departamentoId, actual.getCantidad() - cantidad);
    }

    private Dotacion buscar(int departamentoId) throws NegocioException, SQLException {
        Dotacion t = dotacionDAO.buscarPorDepartamento(departamentoId);
        if (t == null) {
            throw new NegocioException("El departamento no tiene registro de dotación.");
        }
        return t;
    }
}

package pa.ac.unachi.rrhh.presentacion;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.math.BigDecimal;
import java.sql.SQLException;
import pa.ac.unachi.rrhh.modelo.Departamento;
import pa.ac.unachi.rrhh.negocio.DepartamentoService;
import pa.ac.unachi.rrhh.negocio.NegocioException;

/**
 * Controlador de departamentos. Solo lee parámetros, llama al Service
 * y decide la vista. No escribe SQL ni valida reglas de negocio.
 */
@WebServlet(name = "DepartamentoServlet", urlPatterns = {"/departamentos"})
public class DepartamentoServlet extends HttpServlet {

    private final DepartamentoService service = new DepartamentoService();

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String accion = valor(request.getParameter("accion"), "listar");
        try {
            switch (accion) {
                case "nuevo" -> mostrarFormulario(request, response);
                case "editar" -> {
                    int id = Integer.parseInt(request.getParameter("id"));
                    request.setAttribute("departamento", service.buscar(id));
                    mostrarFormulario(request, response);
                }
                default -> listar(request, response);
            }
        } catch (NegocioException | NumberFormatException e) {
            request.setAttribute("error", "No se pudo abrir el departamento solicitado.");
            listarSinError(request, response);
        } catch (SQLException e) {
            throw new ServletException("Error de base de datos", e);
        }
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String accion = valor(request.getParameter("accion"), "");
        try {
            switch (accion) {
                case "guardar" -> guardar(request, response);
                case "eliminar" -> {
                    service.eliminar(Integer.parseInt(request.getParameter("id")));
                    response.sendRedirect(request.getContextPath() + "/departamentos");
                }
                default -> response.sendRedirect(request.getContextPath() + "/departamentos");
            }
        } catch (SQLException e) {
            throw new ServletException("Error de base de datos", e);
        }
    }

    private void guardar(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException, SQLException {
        String idParam = request.getParameter("id");
        String salarioBaseParam = request.getParameter("salarioBase");
        String dotacionMaximaParam = request.getParameter("dotacionMaxima");

        Departamento d = new Departamento();
        d.setCodigo(valor(request.getParameter("codigo"), "").trim().toUpperCase());
        d.setNombre(valor(request.getParameter("nombre"), "").trim());

        // Lectura y conversión con captura explícita de NumberFormatException
        try {
            if (salarioBaseParam != null && !salarioBaseParam.isBlank()) {
                d.setSalarioBase(new BigDecimal(salarioBaseParam.trim()));
            }
            if (dotacionMaximaParam != null && !dotacionMaximaParam.isBlank()) {
                d.setDotacionMaxima(Integer.parseInt(dotacionMaximaParam.trim()));
            }
        } catch (NumberFormatException e) {
            request.setAttribute("error", "Los campos salario base y dotación máxima deben ser valores numéricos válidos.");
            request.setAttribute("departamento", d);
            mostrarFormulario(request, response);
            return;
        }

        try {
            if (idParam == null || idParam.isBlank()) {
                service.crear(d);
            } else {
                d.setId(Integer.parseInt(idParam));
                service.actualizar(d);
            }
            response.sendRedirect(request.getContextPath() + "/departamentos");
        } catch (NegocioException e) {
            // Volver al formulario conservando lo que escribió el usuario
            request.setAttribute("error", e.getMessage());
            request.setAttribute("departamento", d);
            mostrarFormulario(request, response);
        }
    }

    private void listar(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException, SQLException {
        request.setAttribute("departamentos", service.listar());
        request.getRequestDispatcher("/departamentos/listar.jsp").forward(request, response);
    }

    private void listarSinError(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        try {
            listar(request, response);
        } catch (SQLException e) {
            throw new ServletException("Error de base de datos", e);
        }
    }

    private void mostrarFormulario(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        request.getRequestDispatcher("/departamentos/formulario.jsp").forward(request, response);
    }

    private static String valor(String texto, String porDefecto) {
        return texto == null ? porDefecto : texto;
    }
}
